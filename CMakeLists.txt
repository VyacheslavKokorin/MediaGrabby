cmake_minimum_required(VERSION 3.20)
project(MediaGrabby VERSION 0.1.0 LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz)
FetchContent_MakeAvailable(json)
set(BUILD_CURL_EXE OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_LIBCURL_DOCS OFF CACHE BOOL "" FORCE)
set(BUILD_MISC_DOCS OFF CACHE BOOL "" FORCE)
set(HTTP_ONLY ON CACHE BOOL "" FORCE)
set(CURL_USE_SCHANNEL ON CACHE BOOL "" FORCE)
set(CURL_USE_OPENSSL OFF CACHE BOOL "" FORCE)
set(CURL_USE_LIBPSL OFF CACHE BOOL "" FORCE)
set(CURL_ZLIB OFF CACHE BOOL "" FORCE)
set(CURL_BROTLI OFF CACHE BOOL "" FORCE)
set(CURL_ZSTD OFF CACHE BOOL "" FORCE)
set(USE_NGHTTP2 OFF CACHE BOOL "" FORCE)
set(CURL_STATIC_CRT ON CACHE BOOL "" FORCE)
FetchContent_Declare(curl URL https://github.com/curl/curl/releases/download/curl-8_22_0/curl-8.22.0.tar.xz URL_HASH SHA256=f7ef3ae8a22e521f289803fe93543eb64c329b58aa73a9e224dfd915a2a5f4f7)
FetchContent_MakeAvailable(curl)
add_library(mg_platform src/platform.cpp src/proxy.cpp)
target_include_directories(mg_platform PUBLIC src)
target_link_libraries(mg_platform PUBLIC nlohmann_json::nlohmann_json CURL::libcurl bcrypt crypt32 shell32 ole32)
target_compile_definitions(mg_platform PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_executable(MediaGrabby WIN32 src/main.cpp src/terminal.cpp src/appearance.cpp src/proxy_dialog.cpp src/components.cpp src/download.cpp resources/app.rc)
target_link_libraries(MediaGrabby PRIVATE mg_platform comctl32 uxtheme dwmapi comdlg32)
if(MSVC)
  target_compile_options(mg_platform PUBLIC /utf-8 /W4 /EHsc)
  target_link_options(MediaGrabby PRIVATE /MANIFEST:NO)
else()
  target_compile_options(mg_platform PUBLIC -Wall -Wextra)
  target_link_options(MediaGrabby PRIVATE -municode -static)
endif()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_include_directories(core_tests PRIVATE src)
if(MSVC)
 target_compile_options(core_tests PRIVATE /utf-8)
endif()
add_test(NAME core COMMAND core_tests)
add_executable(platform_tests tests/platform_tests.cpp)
target_link_libraries(platform_tests PRIVATE mg_platform)
add_test(NAME platform COMMAND platform_tests)

target_include_directories(MediaGrabby PRIVATE resources)
add_executable(integration_tests tests/integration_tests.cpp src/components.cpp src/download.cpp)
target_link_libraries(integration_tests PRIVATE mg_platform)


add_executable(terminal_tests tests/terminal_tests.cpp src/terminal.cpp)
target_link_libraries(terminal_tests PRIVATE mg_platform comctl32)
add_test(NAME terminal COMMAND terminal_tests)
